// KIRIM DATANG / PULANG DENGAN JAM GLOBAL (MAKASSAR/WITA) async function kirimPresensi(jenisStatus) { const nip = document.getElementById('nip-input').value.trim(); const btnDatang = document.getElementById('btn-datang'); const btnPulang = document.getElementById('btn-pulang'); btnDatang.disabled = true; btnPulang.disabled = true; let tglSekarang = ""; let jamSekarang = ""; try { // Mengambil waktu global secara real-time khusus zona waktu Makassar (WITA) const response = await fetch('https://worldtimeapi.org/api/timezone/Asia/Makassar'); if (!response.ok) throw new Error("Gagal merespons server waktu."); const data = await response.json(); // Memecah hasil datetime dari API (Contoh: "2023-10-27T08:30:00.123+08:00") tglSekarang = data.datetime.split('T')[0]; jamSekarang = data.datetime.split('T')[1].substring(0, 8); // Mengambil HH:MM:SS saja } catch (error) { alert("❌ Gagal mendapatkan waktu global jaringan. Pastikan koneksi internet Anda stabil dan coba lagi."); btnDatang.disabled = false; btnPulang.disabled = false; return; // Membatalkan pengiriman data jika gagal mendapatkan jam global yang valid } const { error } = await db.from('absensi').insert([{ nip: nip, nama: namaPegawai, sekolah: sekolahPegawai, tanggal: tglSekarang, jam: jamSekarang, status: jenisStatus, jarak_m: "0", foto_bukti: fotoBase64 || null }]); if (error) { alert("Gagal mengirim presensi: " + error.message); btnDatang.disabled = false; btnPulang.disabled = false; } else { alert(`✅ Presensi ${jenisStatus} Berhasil Dikirim pada ${tglSekarang} Jam ${jamSekarang} WITA!`); lokasiAwal(); } }
